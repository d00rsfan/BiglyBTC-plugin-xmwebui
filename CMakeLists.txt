cmake_minimum_required(VERSION 3.25)
project(BiglyBTC-plugin-xmwebui VERSION 0.1.0 LANGUAGES CXX)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Toolchain.cmake")
  include(cmake/Toolchain.cmake)
else()
  include(../../cmake/Toolchain.cmake)
endif()
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core)
add_library(biglybtc_rpc_contract STATIC rpc_contract.cpp)
target_include_directories(biglybtc_rpc_contract PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(biglybtc_rpc_contract PUBLIC Qt6::Core)
target_compile_options(biglybtc_rpc_contract PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(biglybtc-rpc-contract-driver contract_driver.cpp)
target_link_libraries(biglybtc-rpc-contract-driver PRIVATE biglybtc_rpc_contract)
target_compile_options(biglybtc-rpc-contract-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME original-xmwebui-rpc-contract COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/fixture_test.py" "$<TARGET_FILE:biglybtc-rpc-contract-driver>")
set_tests_properties(original-xmwebui-rpc-contract PROPERTIES TIMEOUT 90 LABELS "unit;fixtures")
add_executable(biglybtc-rpc-contract-bounds-tests bounds_test.cpp)
target_link_libraries(biglybtc-rpc-contract-bounds-tests PRIVATE biglybtc_rpc_contract)
target_compile_options(biglybtc-rpc-contract-bounds-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bounded-xmwebui-rpc-contract COMMAND biglybtc-rpc-contract-bounds-tests)
set_tests_properties(bounded-xmwebui-rpc-contract PROPERTIES TIMEOUT 30 LABELS "unit")
add_library(biglybtc_rpc_identities STATIC rpc_identities.cpp)
target_include_directories(biglybtc_rpc_identities PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(biglybtc_rpc_identities PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(biglybtc-rpc-identity-driver identity_driver.cpp)
target_link_libraries(biglybtc-rpc-identity-driver PRIVATE biglybtc_rpc_identities Qt6::Core)
target_compile_options(biglybtc-rpc-identity-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME original-xmwebui-rpc-identities COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/identity_fixture_test.py" "$<TARGET_FILE:biglybtc-rpc-identity-driver>")
set_tests_properties(original-xmwebui-rpc-identities PROPERTIES TIMEOUT 90 LABELS "unit;fixtures")
add_executable(biglybtc-rpc-identity-bounds-tests identity_bounds_test.cpp)
target_link_libraries(biglybtc-rpc-identity-bounds-tests PRIVATE biglybtc_rpc_identities)
target_compile_options(biglybtc-rpc-identity-bounds-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bounded-xmwebui-rpc-identities COMMAND biglybtc-rpc-identity-bounds-tests)
set_tests_properties(bounded-xmwebui-rpc-identities PROPERTIES TIMEOUT 30 LABELS "unit")
